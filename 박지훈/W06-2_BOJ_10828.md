## boj 10828번 문제
https://www.acmicpc.net/problem/10828

```
from collections import deque
import sys

input = sys.stdin.readline

orders = int(input())
stk = []
stack = deque(stk)

for _ in range(orders):
    order, *num = input().split()
    if order.startswith("push"):
        stack.appendleft(num[0])
    
    if order == "pop":
        if len(stack) == 0:
            print(-1)
        else:
            number = stack.popleft()
            print(number)
    
    if order == "top":
        if len(stack) == 0:
            print(-1)
        else:
            print(stack[0])
    
    if order == "size":
        print(len(stack))
    
    if order == "empty":
        if len(stack) > 0:
            print(0)
        else:
            print(1)
```
