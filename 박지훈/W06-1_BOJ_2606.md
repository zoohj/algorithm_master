## boj 2606번 문제
https://www.acmicpc.net/problem/2606

```
from collections import deque
import sys

num_of_pc = int(sys.stdin.readline())
conenct_pc = int(sys.stdin.readline())
Relationship_pcs = []
Infectious_Agent = [1]

result_set = set(Infectious_Agent)
queue = deque(Infectious_Agent)

for i in range(conenct_pc):
    Relationship_pc = list(map(int, sys.stdin.readline().split()))
    Relationship_pcs.append(Relationship_pc)

while queue:
    current = queue.popleft()  # 현재 기준이 될 숫자 하나 꺼내기
    
    for pair in Relationship_pcs:
        # 서브 리스트에 현재 숫자가 포함되어 있는지 확인
        if current in pair:
            # 두 숫자 중 현재 숫자가 아닌 "다른 숫자" 찾기
            next_val = pair[1] if pair[0] == current else pair[0]
            
            # 그 숫자가 아직 결과 셋에 없다면 새로 추가!
            if next_val not in result_set:
                result_set.add(next_val)
                queue.append(next_val) # 새로 찾은 숫자로 또 친구 찾기 위해 큐에 넣음

print(len(result_set) - 1)
```
