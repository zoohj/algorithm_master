## boj 2805번 문제
https://www.acmicpc.net/problem/2805

```
import sys

# 1. 입력 받기 (sys.stdin.readline은 대량 데이터 입력 시 필수!)
N, M = map(int, sys.stdin.readline().split())
trees = list(map(int, sys.stdin.readline().split()))

# 2. 이진 탐색을 위한 시작점과 끝점 설정
start = 0
end = max(trees)

answer = 0

# 3. 이진 탐색 시작
while start <= end:
    mid = (start + end) // 2  # 현재 설정한 절단기 높이
    
    # mid 높이로 잘랐을 때 가져갈 수 있는 나무의 총합 계산
    # 리스트를 새로 만들지 않고 바로 합산해야 속도가 빠릅니다.
    total = 0
    for tree in trees:
        if tree > mid:
            total += tree - mid
            
    # 4. 가져가는 나무 양에 따라 범위 좁히기
    if total >= M:
        # 나무가 충분하거나 남는 경우 -> 절단기 높이를 더 높여보자!
        answer = mid  # 일단 현재 높이를 답으로 저장
        start = mid + 1
    else:
        # 나무가 모자란 경우 -> 절단기 높이를 낮춰야 함
        end = mid - 1

# 5. 최적의 절단기 높이 출력
print(answer)
```
